// FURNISPACE DATABASE SCHEMA - DBML v2
// FurniSpace là hệ thống nghiệp vụ của doanh nghiệp cung cấp
// giải pháp nội thất / thiết kế không gian bán lẻ.
// Ghi chú:
// - Không dùng consultation_requests riêng, dùng projects.status để quản lý từ request đến hoàn tất.
// - Không tách role Delivery Staff; Sales/Production phối hợp xử lý giao hàng.
// - Không dùng bảng deliveries / delivery_items; giao hàng được tracking qua orders, order_items và project_schedules.
// - Dữ liệu scene 2D/3D chi tiết lưu ở MongoDB, SQL chỉ lưu metadata và business items.
// - File lưu qua files + file_links.

// =========================================================
// 1. ACCOUNT & ROLE MANAGEMENT
// =========================================================

// Bảng vai trò trong hệ thống.
// Ví dụ: CUSTOMER, SALES_CONSULTANT, DESIGNER_STAFF,
// PRODUCTION_STAFF, ADMIN.
Table roles {
  role_id uuid [pk]
  role_name varchar(50) [unique, not null]
  description text

  created_at timestamp
  updated_at timestamp
}

// Bảng tài khoản người dùng.
// Hiện tại một account có một role chính thông qua role_id.
// Nếu sau này muốn một người có nhiều role thì thêm bảng account_roles.
Table accounts {
  account_id uuid [pk]

  role_id uuid [not null, ref: > roles.role_id]

  email varchar(100) [unique, not null]
  password varchar(255) [not null]
  full_name nvarchar(100) [not null]
  phone varchar(20)
  avatar_url text

  status account_status [default: 'ACTIVE']

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}


// =========================================================
// 2. PRODUCT MANAGEMENT
// =========================================================

// Danh mục sản phẩm.
// Ví dụ: Bàn, Ghế, Tủ trưng bày, Kệ, Quầy thu ngân.
Table categories {
  category_id uuid [pk]



  category_name varchar(100) [not null]
  description text
  status product_status [default: 'ACTIVE']

  created_at timestamp
  updated_at timestamp
}

// Sản phẩm tổng quát trong catalog.
// Ví dụ: Display Cabinet, Coffee Table, Cashier Counter.
Table products {
  product_id uuid [pk]

  category_id uuid [ref: > categories.category_id]

  product_code varchar(50) [unique]
  product_name varchar(150) [not null]
  description text

  status product_status [default: 'ACTIVE']

  created_at timestamp
  updated_at timestamp
}

// Phiên bản cụ thể của sản phẩm.
// Ví dụ: Cabinet 4 ngăn màu nâu, Cabinet 5 ngăn màu trắng.
// Nếu là phiên bản riêng cho project thì project_id sẽ có giá trị.
Table product_versions {
  product_version_id uuid [pk]

  product_id uuid [not null, ref: > products.product_id]

  // THÊM
  project_id uuid [ref: > projects.project_id]  // Thêm để truy vấn: Product Version custom cho Project A
                                                                  // → is_project_specific = true
                                                                  // → is_public = false
                                                                  // → project_id = project_id của Project A
  
  dimension_unit varchar(10) [default: 'cm']  // Room Planner cần biết width/height/depth đang là cm, m hay mm. Nếu không, Babylon.js dễ scale sai model.

  version_code varchar(50) [unique]
  version_name varchar(150) [not null]
  version_type product_version_type [default: 'STANDARD']

  material varchar(100)
  color varchar(100)

  width decimal(10,2)
  height decimal(10,2)
  depth decimal(10,2)

  estimated_price decimal(12,2)

  is_default boolean [default: false]
  is_public boolean [default: true] // false nếu không muốn hiện cho project khác
  is_project_specific boolean [default: false]

  status product_status [default: 'ACTIVE']

  created_at timestamp
  updated_at timestamp
}


// =========================================================
// 3. GENERIC FILE STORAGE
// =========================================================

// Bảng lưu thông tin file vật lý.
// Bảng này chỉ lưu metadata của file upload.
// File thuộc object nào sẽ được lưu ở bảng file_links.
Table files {
  file_id uuid [pk]                         // Khóa chính của file

  uploaded_by uuid [not null, ref: > accounts.account_id] 
  // Người upload file lên hệ thống

  original_file_name varchar(255) [not null]
  // Tên file gốc do user upload, ví dụ: "mat-bang-quan-cafe.pdf"

  stored_file_name varchar(255) [not null]
  // Tên file đã được hệ thống đổi để tránh trùng, ví dụ: "uuid-current-space.jpg"

  file_url text [not null]
  // URL dùng để xem hoặc tải file từ Firebase Storage

  storage_path text [not null]
  // Đường dẫn thật trong Firebase Storage, ví dụ: "projects/{projectId}/files/{fileName}"

  mime_type varchar(100) [not null]
  // MIME type của file, ví dụ: image/jpeg, application/pdf, model/gltf-binary

  file_extension varchar(20)
  // Phần mở rộng của file, ví dụ: jpg, png, pdf, glb

  file_size_bytes bigint [not null]
  // Dung lượng file tính bằng bytes

  checksum varchar(255)
  // Mã kiểm tra file, dùng để phát hiện file trùng hoặc kiểm tra toàn vẹn file nếu cần

  status file_status [not null, default: 'ACTIVE']
  // Trạng thái file: ACTIVE, ARCHIVED, DELETED

  uploaded_at timestamp [not null]
  // Thời điểm file được upload

  archived_at timestamp
  // Thời điểm file được archive, nếu có


  indexes {
  uploaded_by [name: 'idx_files_uploaded_by']
  status [name: 'idx_files_status']
  storage_path [unique, name: 'uq_files_storage_path']
  }
}

// Bảng link file tới các object nghiệp vụ.
// Một file có thể gắn với nhiều object khác nhau.
//
// owner_type ví dụ:
// PROJECT, PROJECT_AREA, PROJECT_SCHEDULE, PROPOSAL,
// PROPOSAL_SCENE, PRODUCT_VERSION, CUSTOMIZATION_REQUEST,
// QUOTATION, ORDER, PRODUCTION_REQUEST, PROJECT_REVIEW.
//
// Lưu ý: owner_id là polymorphic reference.
// Database không tự enforce FK cho owner_id.
// Backend/service layer phải validate owner_type + owner_id.
Table file_links {
  file_link_id uuid [pk]   // Khóa chính của liên kết file

  file_id uuid [not null, ref: > files.file_id]
  // File được liên kết

  reference_type varchar(50) [not null]
  // Loại đối tượng nghiệp vụ mà file gắn vào, ví dụ: PROJECT, PROPOSAL, ORDER, PROJECT_SCHEDULE

  reference_id uuid [not null]
  // ID của đối tượng nghiệp vụ tương ứng với reference_type

  file_type file_type [not null, default: "OTHER"]
  // Loại file theo nghiệp vụ, ví dụ: SPACE_IMAGE, FLOOR_PLAN, MEASUREMENT_FILE, DESIGN_PREVIEW, CONTRACT, OTHER

  visibility file_visibility [not null, default: 'CUSTOMER_VISIBLE']
  // Quyền hiển thị của file

  // Thêm
  is_primary boolean [default: false]
  // UI cần biết ảnh nào là ảnh chính

  display_order int [default: 0]
  // Kéo thả thứ tự sản phẩm 

  description text
  // Mô tả ngắn về file

  created_by uuid [ref: > accounts.account_id]
  // Người tạo liên kết file với đối tượng nghiệp vụ

  created_at timestamp [not null]
  // Thời điểm tạo liên kết file

  indexes {
    (reference_type, reference_id) [name: 'idx_file_links_reference']
    file_id [name: 'idx_file_links_file_id']
    (file_id, reference_type, reference_id, file_type) [unique, name: 'uq_file_links_unique_reference']
  }
}

// Bảng project trung tâm.
// Project được tạo từ giai đoạn customer request ban đầu,
// không dùng bảng consultation_requests riêng.
// Toàn bộ flow được điều khiển bằng project.status.
Table projects {
  project_id uuid [pk]

  customer_id uuid [not null, ref: > accounts.account_id]

  assigned_sales_id uuid [ref: > accounts.account_id]     // Sales hiện tại đang phụ trách
  assigned_designer_id uuid [ref: > accounts.account_id]  // Designer hiện tại đang phụ trách

  project_code varchar(50) [unique]
  project_name varchar(150) [not null]

  business_type varchar(100)       // Loại cửa hàng: cafe, fashion store, showroom...
  project_address text
  business_purpose text            // Optional

  // Mô tả tự nhiên của khách về nội thất cần thiết.
  // Ví dụ: "4 bàn tròn khoảng 2m, ghế 4 chân có tựa lưng,
  // quầy thu ngân nhỏ, vài kệ trưng bày..."
  furniture_requirement text

  description text

  total_area_sqm decimal(10,2)
  number_of_floors int

  budget_min decimal(12,2)          // Optional
  budget_max decimal(12,2)          // Optional

  target_completion_date date // Ngày khách mong muốn hoàn tất project, ví dụ trước ngày khai trương

  status project_status [default: 'SUBMITTED']

  submitted_at timestamp
  sales_assigned_at timestamp
  approved_at timestamp
  designer_assigned_at timestamp

  completed_at timestamp
  rejected_at timestamp

  rejection_reason text

  created_at timestamp
  updated_at timestamp
}

// Lịch sử phân công nhân sự cho project.
// Dùng để biết ai từng phụ trách Sales/Designer/Production.

// Thông báo workflow cho người dùng.
// Ví dụ: proposal đã sẵn sàng, báo giá đã gửi, lịch đo/lịch giao đã tạo.
Table notifications {
  notification_id uuid [pk]
  // Khóa chính của notification

  receiver_id uuid [not null, ref: > accounts.account_id]
  // Người nhận notification

  project_id uuid [ref: > projects.project_id]
  // Project liên quan, dùng để FE có context và fallback route về project detail

  title varchar(150) [not null]
  // Tiêu đề thông báo

  message text
  // Nội dung thông báo đã được render sẵn từ template

  notification_type varchar(50)
  // Loại notification/event, ví dụ:
  // PROJECT_REQUEST_SUBMITTED
  // PROJECT_REQUEST_ACCEPTED
  // PROJECT_MORE_INFORMATION_REQUESTED
  // PROJECT_BASIC_INFORMATION_UPDATED
  // PROJECT_STATUS_CHANGED
  // PROJECT_REQUEST_REJECTED
  // PROJECT_DESIGNER_ASSIGNED
  // PROJECT_FILE_UPLOADED
  // PROJECT_SCHEDULE_CREATED
  // PROJECT_SCHEDULE_UPDATED
  // PROJECT_SCHEDULE_CONFIRMED
  // PROJECT_SCHEDULE_CANCELLED

  reference_type varchar(50)
  // Loại object mà notification trỏ tới, ví dụ:
  // PROJECT, PROJECT_SCHEDULE, PROPOSAL, QUOTATION, ORDER, FILE

  reference_id uuid
  // ID của object theo reference_type.
  // Ví dụ reference_type = PROJECT_SCHEDULE thì reference_id = schedule_id

  is_read boolean [not null, default: false]
  // Đã đọc hay chưa

  read_at timestamp
  // Thời điểm user đọc notification

  deleted_at timestamp
  // Soft delete notification.
  // Khi query danh sách notification thì lọc deleted_at IS NULL

  created_at timestamp [not null]
  // Thời điểm tạo notification

  indexes {
    (receiver_id, is_read) [name: 'idx_notifications_receiver_read']
    (receiver_id, created_at) [name: 'idx_notifications_receiver_created']
    (reference_type, reference_id) [name: 'idx_notifications_reference']
    project_id [name: 'idx_notifications_project_id']
  }

// Khoản thanh toán / payment summary.

Table payments {
  payment_id uuid [pk] 
  // Khóa chính của khoản thanh toán

  project_id uuid [not null, ref: > projects.project_id]
  // Dự án liên quan đến khoản thanh toán

  // Đơn hàng liên quan đến khoản thanh toán, nếu có

  // Báo giá liên quan đến khoản thanh toán, nếu có

  paid_by uuid [ref: > accounts.account_id]
  // Người cần thanh toán, thường là Customer

  payment_type payment_type [default: 'OTHER']
  // Loại khoản thanh toán: DEPOSIT, REMAINING_PAYMENT, FULL_PAYMENT, MEASUREMENT_FEE, DESIGN_FEE, CUSTOMIZATION_FEE, DELIVERY_FEE, CANCELLATION_FEE, REFUND, OTHER

  amount decimal(12,2) [not null]
  // Số tiền cần thanh toán của khoản này

  remaining_amount decimal(12,2) [not null, default: 0]
  // Số tiền còn lại cần thanh toán

  currency varchar(10) [not null, default: 'VND']
  // Đơn vị tiền tệ

  status payment_status [default: 'PENDING']
  // Trạng thái tổng hiện tại của payment

  due_date date
  // Hạn thanh toán nếu có

  paid_at timestamp
  // Thời điểm khoản thanh toán được xem là đã thanh toán đủ

  cancelled_at timestamp
  // Thời điểm khoản thanh toán bị hủy, nếu có

  note text
  // Ghi chú nội bộ về khoản thanh toán

  created_at timestamp
  // Thời điểm tạo payment

  updated_at timestamp
  // Thời điểm cập nhật payment gần nhất
}

Table payment_transactions {
  payment_transaction_id uuid [pk]
  // Khóa chính của transaction

  payment_id uuid [not null, ref: > payments.payment_id]
  // Khoản payment mà transaction này thuộc về

  project_id uuid [ref: > projects.project_id]
  // Dự án liên quan, lưu thêm để query lịch sử giao dịch theo project nhanh hơn

  order_id uuid [ref: > orders.order_id]
  // Đơn hàng liên quan, lưu thêm để query lịch sử giao dịch theo order nhanh hơn

  transaction_code varchar(100) [not null, unique]
  // Mã giao dịch nội bộ, ví dụ: TXN-2026-0001

  transaction_type payment_transaction_type [not null, default: 'CHARGE']
  // Loại giao dịch: CHARGE, REFUND, ADJUSTMENT

  amount decimal(12,2) [not null]
  // Số tiền của giao dịch này

  currency varchar(10) [not null, default: 'VND']
  // Đơn vị tiền tệ

  payment_method varchar(50)
  // Phương thức thanh toán, ví dụ: CASH, BANK_TRANSFER, VNPAY, MOMO

  provider_transaction_id varchar(255)
  // Mã giao dịch từ cổng thanh toán nếu có

  status payment_transaction_status [not null, default: 'PENDING']
  // Trạng thái của giao dịch

  transaction_time timestamp
  // Thời điểm giao dịch thực tế xảy ra

  confirmed_by uuid [ref: > accounts.account_id]
  // Nhân viên xác nhận giao dịch nếu cần xác nhận thủ công

  confirmed_at timestamp
  // Thời điểm xác nhận giao dịch

  failure_reason text
  // Lý do thất bại nếu giao dịch thất bại

  raw_provider_payload jsonb
  // Dữ liệu callback/raw response từ payment gateway hoặc sao kê gốc nếu cần lưu

  created_at timestamp [not null]
  // Thời điểm tạo transaction record

  indexes {
    payment_id [name: 'idx_payment_transactions_payment_id']
    (project_id, created_at) [name: 'idx_payment_transactions_project_time']
    (order_id, created_at) [name: 'idx_payment_transactions_order_time']
    provider_transaction_id [name: 'idx_payment_transactions_provider_transaction_id']
  }
}

// =========================================================
// 14. PROPOSAL SCENE VARIANT MANAGEMENT
// Dùng khi Customer được phép thử/suggest thay đổi trên bản vẽ:
// - đổi vị trí product
// - đổi màu/vật liệu
// - đổi sang Product Version khác
// nhưng không được sửa trực tiếp bản official của Designer.
// =========================================================

// Trạng thái của một scene variant.
// DRAFT: Customer/Designer đang chỉnh, chưa gửi.
// SUBMITTED: đã gửi cho Designer review.
// ACCEPTED: Designer đồng ý suggestion.
// REJECTED: Designer từ chối suggestion.
// APPLIED: suggestion đã được apply vào official proposal scene.
Enum proposal_scene_variant_status {
  DRAFT
  SUBMITTED
  ACCEPTED
  REJECTED
  APPLIED
}

// Loại variant.
// CUSTOMER_SUGGESTION: bản Customer tự chỉnh để góp ý.
// DESIGNER_REVISION: bản Designer tạo như một revision/alternative.
Enum proposal_scene_variant_type {
  CUSTOMER_SUGGESTION
  DESIGNER_REVISION
}

// Bảng metadata cho các bản scene variant.
// Dữ liệu 3D chi tiết của variant vẫn lưu MongoDB qua mongo_variant_scene_id.
// SQL chỉ lưu metadata để query, phân quyền, review và tracking trạng thái.
Table proposal_scene_variants {
  variant_id uuid [pk]
  // Khóa chính của scene variant.

  proposal_id uuid [not null, ref: > proposals.proposal_id]
  // Proposal mà variant này thuộc về.
  // Giúp query tất cả variant theo một proposal.

  scene_id uuid [not null, ref: > proposal_scenes.scene_id]
  // Official scene gốc mà variant dựa trên.
  // Variant không thay thế trực tiếp scene này cho tới khi Designer apply.

  created_by uuid [not null, ref: > accounts.account_id]
  // Người tạo variant.
  // Thường là Customer nếu variant_type = CUSTOMER_SUGGESTION.
  // Có thể là Designer nếu variant_type = DESIGNER_REVISION.

  variant_type proposal_scene_variant_type [default: 'CUSTOMER_SUGGESTION']
  // Phân biệt variant do Customer suggest hay Designer tạo revision.

  status proposal_scene_variant_status [default: 'DRAFT']
  // Trạng thái review/apply của variant.

  mongo_variant_scene_id varchar(100) [not null]
  // ID document trong MongoDB lưu scene data của variant.
  // Document này lưu object positions, rotations, material override,...
  // Không lưu full JSON scene trong SQL.

  note text
  // Ghi chú tổng quan của người tạo variant.
  // Ví dụ: "Muốn đổi ghế sang màu nâu và kéo bàn sát cửa sổ hơn."

  submitted_at timestamp
  // Thời điểm Customer/Designer submit variant để review.
  // Null nếu variant còn DRAFT.

  reviewed_by uuid [ref: > accounts.account_id]
  // Người review variant.
  // Thường là assigned Designer hoặc Admin.

  reviewed_at timestamp
  // Thời điểm Designer/Admin review variant.

  review_note text
  // Ghi chú khi Designer accept/reject.
  // Ví dụ: "Đồng ý đổi màu ghế nhưng không nên kéo bàn sát cửa vì vướng lối đi."

  applied_at timestamp
  // Thời điểm variant được apply vào official scene.
  // Chỉ có giá trị khi status = APPLIED.

  applied_by uuid [ref: > accounts.account_id]
